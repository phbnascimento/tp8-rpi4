start_file=start4.elf
fixup_file=fixup4.dat

kernel=Image

# To use an external initramfs file
#initramfs rootfs.cpio.gz

# Disable overscan assuming the display supports displaying the full resolution
# If the text shown on the screen disappears off the edge, comment this out
disable_overscan=1

# How much memory in MB to assign to the GPU on Pi models having
# 256, 512 or 1024 MB total memory
gpu_mem_256=100
gpu_mem_512=100
gpu_mem_1024=100

# Enable UART0 for serial console on ttyAMA0
dtoverlay=miniuart-bt

# enable 64bits support
arm_64bit=1

# Carrega o overlay do HC-SR04 (hcsr04.dtbo, na pasta overlays/ desta
# partição). Sem esta linha o kernel não cria o nó do sensor, o probe() do
# driver nunca roda e /dev/hcsr04 não existe.
dtoverlay=hcsr04

# --- Tempo de boot ----------------------------------------------------
# disable_splash: não desenha o arco-íris antes de entregar o controle.
# boot_delay: o firmware não espera antes de carregar o kernel.
# initial_turbo foi testado e REMOVIDO: ele força clock máximo nos primeiros
#   30 s, aumentando o consumo enquanto a fonte ainda estabiliza. Numa placa
#   real apareceram teclado USB falhando e HDMI em resolução errada — sintomas
#   clássicos de alimentação marginal. O ganho era ~1 s, estimado e nunca
#   medido; não compensa o risco no dia da apresentação.
disable_splash=1
boot_delay=0
